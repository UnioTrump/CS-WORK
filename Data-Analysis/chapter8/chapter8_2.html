# ---- 5a. 单层（相当于逻辑回归，没有隐藏层）----
print("正在训练 单层模型（无隐藏层）...")
mlp_single = MLPClassifier(
    hidden_layer_sizes=(),          # 无隐藏层 = 单神经元
    activation='relu',
    max_iter=500,
    random_state=42
)
mlp_single.fit(X_train, y_train)
y_pred_single = mlp_single.predict(X_test)
acc_single = accuracy_score(y_test, y_pred_single)
print(f"   单层模型准确率: {acc_single:.4f}")

# ---- 5b. 浅层（1个隐藏层，3个神经元）----
print("正在训练 浅层模型（1个隐藏层，3个神经元）...")
mlp_shallow = MLPClassifier(
    hidden_layer_sizes=(3,),        # 1个隐藏层，3个神经元
    activation='relu',
    max_iter=500,
    random_state=42
)
mlp_shallow.fit(X_train, y_train)
y_pred_shallow = mlp_shallow.predict(X_test)
acc_shallow = accuracy_score(y_test, y_pred_shallow)
print(f"   浅层模型准确率: {acc_shallow:.4f}")

# ---- 5c. 深层（2个隐藏层，每层10个神经元）----
print("正在训练 深层模型（2个隐藏层，每层10个神经元）...")
mlp_deep = MLPClassifier(
    hidden_layer_sizes=(10, 10, 10),    # 2个隐藏层，各10个神经元
    activation='tanh',
    max_iter=500,
    random_state=42
)
mlp_deep.fit(X_train, y_train)
y_pred_deep = mlp_deep.predict(X_test)
acc_deep = accuracy_score(y_test, y_pred_deep)
print(f"   深层模型准确率: {acc_deep:.4f}")

def plot_decision_boundary(model, X, y, ax, title):
    """画出分类模型的决策边界"""
    x_min, x_max = X[:, 0].min() - 0.5, X[:, 0].max() + 0.5
    y_min, y_max = X[:, 1].min() - 0.5, X[:, 1].max() + 0.5
    xx, yy = np.meshgrid(np.linspace(x_min, x_max, 200),
                         np.linspace(y_min, y_max, 200))

    # 预测网格上每个点的类别
    Z = model.predict(np.c_[xx.ravel(), yy.ravel()])
    Z = Z.reshape(xx.shape)

    # 填充颜色区域 = 决策边界
    ax.contourf(xx, yy, Z, alpha=0.3, cmap='coolwarm')
    # 画原始数据点
    ax.scatter(X[y==0, 0], X[y==0, 1], c='red', edgecolors='k', s=30)
    ax.scatter(X[y==1, 0], X[y==1, 1], c='blue', edgecolors='k', s=30)
    ax.set_title(title)
    ax.grid(True, alpha=0.3)
