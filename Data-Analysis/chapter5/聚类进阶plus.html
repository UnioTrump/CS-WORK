import numpy as np
import matplotlib.pyplot as plt

from sklearn.datasets import make_blobs
from sklearn.preprocessing import StandardScaler
from sklearn.cluster import AgglomerativeClustering, Birch, MeanShift, estimate_bandwidth
from sklearn.metrics import silhouette_score, adjusted_rand_score

RANDOM_STATE = 42

def build_dataset(n_samples=900):
    """
    构造三簇数据，各簇方差不同，并施加线性变换制造拉伸效果。
    返回标准化后的 X 和真实标签 y_true（用于计算 ARI）。
    """
    centers = [(-2.5, -1.0), (2.5, 0.0), (0.0, 3.0)]
    stds = [0.55, 1.05, 0.75]
    X, y_true = make_blobs(
        n_samples=n_samples,
        centers=centers,
        cluster_std=stds,
        random_state=RANDOM_STATE,
    )

    # 非刚性变换，使簇呈非球形（测试算法对形状的适应性）
    transform = np.array([[0.75, -0.40], [0.25, 0.95]])
    X = np.dot(X, transform)

    X = StandardScaler().fit_transform(X)
    return X, y_true


def safe_internal_score(X, labels):
    """当有效簇数少于2时返回 NaN，避免 silhouette_score 报错。"""
    unique = np.unique(labels)
    if len(unique) < 2:
        return np.nan
    return silhouette_score(X, labels)


def evaluate(name, X, y_true, labels):
    """计算聚类结果的两个评估指标：轮廓系数（内部）和 ARI（外部）。"""
    sil = safe_internal_score(X, labels)
    ari = adjusted_rand_score(y_true, labels)
    n_clusters = len(np.unique(labels))
    return {
        "name": name,
        "labels": labels,
        "n_clusters": n_clusters,
        "silhouette": sil,
        "ari": ari,
    }


def run_models(X, y_true):
    """运行所有待比较的聚类算法，返回结果列表。"""
    results = []

    # 层次聚类：对比不同 linkage（ward / complete / average）
    for linkage in ["ward", "complete", "average"]:
        model = AgglomerativeClustering(n_clusters=3, linkage=linkage)
        labels = model.fit_predict(X)
        results.append(evaluate("Agglomerative-" + linkage, X, y_true, labels))

    # Birch：使用阈值控制子簇合并，适合大规模数据
    birch = Birch(n_clusters=3, threshold=0.35)
    labels_b = birch.fit_predict(X)
    results.append(evaluate("Birch(th=0.35)", X, y_true, labels_b))

    # MeanShift：自动确定簇数，带宽估计采用 quantile 方法
    bandwidth = estimate_bandwidth(X, quantile=0.20, n_samples=min(600, X.shape[0]))
    if bandwidth <= 0:          # 鲁棒性处理：避免 bandwidth 为 0 或负值
        bandwidth = 0.8
    ms = MeanShift(bandwidth=bandwidth, bin_seeding=True)
    labels_m = ms.fit_predict(X)
    results.append(evaluate("MeanShift", X, y_true, labels_m))

    return results


def print_table(results):
    """打印带边框的对比表格，包含簇数、轮廓系数、ARI。"""
    header = "{:<24} | {:>9} | {:>10} | {:>10}".format(
        "Model", "#Clusters", "Silhouette", "ARI"
    )
    print("\n" + header)
    print("-" * len(header))
    for r in results:
        print(
            "{:<24} | {:>9d} | {:>10.4f} | {:>10.4f}".format(
                r["name"], r["n_clusters"], r["silhouette"], r["ari"]
            )
        )


def plot_all(X, y_true, results):
    """绘制真实标签和各模型的聚类结果散点图，便于直观对比。"""
    total = len(results) + 1
    ncols = 3
    nrows = int(np.ceil(total / float(ncols)))

    fig, axes = plt.subplots(nrows, ncols, figsize=(5 * ncols, 4 * nrows))
    axes = np.array(axes).reshape(-1)

    # 第一子图：真实标签
    ax0 = axes[0]
    ax0.scatter(X[:, 0], X[:, 1], c=y_true, s=16, cmap="tab10", alpha=0.85)
    ax0.set_title("Ground Truth")
    ax0.set_xlabel("x1")
    ax0.set_ylabel("x2")
    ax0.grid(alpha=0.2)

    # 其余子图：每个模型的结果
    for i, r in enumerate(results, start=1):
        ax = axes[i]
        ax.scatter(X[:, 0], X[:, 1], c=r["labels"], s=16, cmap="tab10", alpha=0.85)
        title = "{}\nSil={:.3f}, ARI={:.3f}".format(r["name"], r["silhouette"], r["ari"])
        ax.set_title(title)
        ax.set_xlabel("x1")
        ax.set_ylabel("x2")
        ax.grid(alpha=0.2)

    # 删除多余子图（当子图数量多于实际模型数+1时）
    for j in range(total, len(axes)):
        fig.delaxes(axes[j])

    plt.tight_layout()
    plt.show()

def main():
    X, y_true = build_dataset(n_samples=900)
    results = run_models(X, y_true)
    print_table(results)
    plot_all(X, y_true, results)


if __name__ == "__main__":
    main()
